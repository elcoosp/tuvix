+//! Benchmarks for the M11 interaction primitives: the keymap matcher and
+//! the focus ring. The input-dispatch budget these back is the B10
+//! input-to-paint pipeline — a keymap step or a focus cycle must stay a
+//! rounding error next to the ~295 us p99 that budget measured.
+
+use std::hint::black_box;
+
+use criterion::{Criterion, Throughput, criterion_group, criterion_main};
+use tuvix_core::event::{KeyCode, KeyEvent, KeyModifiers};
+use tuvix_core::focus::FocusRing;
+use tuvix_core::keymap::{KeyPattern, Keymap};
+
+#[derive(Debug, Clone, Copy, PartialEq, Eq)]
+enum Command {
+    Quit,
+    GotoTop,
+    DeleteWord,
+    Save,
+    Next,
+    Previous,
+}
+
+fn dashboard_keymap() -> Keymap<Command> {
+    let mut keymap = Keymap::new();
+    // Singles.
+    keymap.bind(&[KeyPattern::new(KeyCode::Char('c')).ctrl()], Command::Quit);
+    keymap.bind(&[KeyPattern::new(KeyCode::Char('s')).ctrl()], Command::Save);
+    keymap.bind(&[KeyPattern::new(KeyCode::Down)], Command::Next);
+    keymap.bind(&[KeyPattern::new(KeyCode::Up)], Command::Previous);
+    // Chords sharing a prefix.
+    keymap.bind(
+        &[
+            KeyPattern::new(KeyCode::Char('g')),
+            KeyPattern::new(KeyCode::Char('g')),
+        ],
+        Command::GotoTop,
+    );
+    keymap.bind(
+        &[
+            KeyPattern::new(KeyCode::Char('d')),
+            KeyPattern::new(KeyCode::Char('d')),
+        ],
+        Command::DeleteWord,
+    );
+    keymap
+}
+
+fn bench_interaction_keymap(c: &mut Criterion) {
+    let mut group = c.benchmark_group("interaction");
+    group.throughput(Throughput::Elements(1));
+
+    group.bench_function("keymap_single_key_step", |b| {
+        let mut keymap = dashboard_keymap();
+        let down = KeyEvent::new(KeyCode::Down, KeyModifiers::empty());
+        b.iter(|| {
+            black_box(keymap.step(black_box(&down)));
+        });
+    });
+
+    // The chord walk: the two steps alternate between opening the prefix
+    // (Pending) and completing it (Matched + reset).
+    group.bench_function("keymap_chord_prefix_step", |b| {
+        let mut keymap = dashboard_keymap();
+        let g = KeyEvent::new(KeyCode::Char('g'), KeyModifiers::empty());
+        b.iter(|| {
+            let outcome = keymap.step(black_box(&g));
+            black_box(outcome);
+        });
+    });
+
+    // The worst case at the root: an unbound key scanning every sibling.
+    group.bench_function("keymap_miss_step", |b| {
+        let mut keymap = dashboard_keymap();
+        let x = KeyEvent::new(KeyCode::Char('x'), KeyModifiers::empty());
+        b.iter(|| {
+            black_box(keymap.step(black_box(&x)));
+        });
+    });
+
+    group.finish();
+}

+fn bench_interaction_focus(c: &mut Criterion) {
+    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
+    enum Field {
+        A,
+        B,
+        C,
+        D,
+        E,
+        F,
+        G,
+        H,
+    }
+
+    let mut group = c.benchmark_group("interaction");
+    group.throughput(Throughput::Elements(1));
+

+//! The focus ring: which widget receives the editing keys.
+//!
+//! tuvix widget states stay key-agnostic — the app translates events into
+//! state methods (the M6 `Input` contract). What apps still hand-roll is
+//! the *routing*: "which of my three fields is focused, and where does Tab
+//! go next?" [`FocusRing`] is that routing as a type: an ordered ring of
+//! focus targets with wrap-around cycling, lookup by position or by value,
+//! and an explicit unfocused state.
+//!
+//! The ring is generic: focus targets are whatever the app names its
+//! fields with — an enum of widgets, `usize` indices, `&'static str` tags.
+//!
+//! # Examples
+//!
+//! ```
+//! use tuvix_core::focus::FocusRing;
+//!
+//! #[derive(Debug, Clone, Copy, PartialEq, Eq)]
+//! enum Field {
+//!     Name,
+//!     Email,
+//!     Bio,
+//! }
+//!
+//! let mut ring = FocusRing::with_items([Field::Name, Field::Email, Field::Bio]);
+//! assert!(ring.is_focused(&Field::Name));
+//! ring.advance();
+//! assert!(ring.is_focused(&Field::Email));
+//! ring.advance();
+//! ring.advance(); // wraps
+//! assert!(ring.is_focused(&Field::Name));
+//! assert!(ring.set(&Field::Bio));
+//! assert_eq!(ring.current(), Some(&Field::Bio));
+//! ```
+
+/// An ordered ring of focus targets with wrap-around cycling.
+///
+/// The ring starts focused on its first item (or unfocused when empty);
+/// [`FocusRing::unfocus`] clears the focus explicitly, and the next cycle
+/// restarts from the beginning.
+#[derive(Debug, Clone, PartialEq, Eq)]
+pub struct FocusRing<T> {
+    items: Vec<T>,
+    index: Option<usize>,
+}
+
+impl<T> Default for FocusRing<T> {
+    /// An empty, unfocused ring — `T` needs no default of its own.
+    fn default() -> Self {
+        Self::new()
+    }
+}
+
+impl<T> FocusRing<T> {
+    /// Creates an empty, unfocused ring.
+    pub fn new() -> Self {
+        Self {
+            items: Vec::new(),
+            index: None,
+        }
+    }
+
+    /// Creates a ring over `items`, focused on the first one.
+    pub fn with_items(items: impl IntoIterator<Item = T>) -> Self {
+        let items: Vec<T> = items.into_iter().collect();
+        let index = if items.is_empty() { None } else { Some(0) };
+        Self { items, index }
+    }
+
+    /// Appends a target; an empty ring focuses it.
+    pub fn push(&mut self, item: T) {
+        let was_empty = self.items.is_empty();
+        self.items.push(item);
+        if was_empty {
+            self.index = Some(0);
+        }
+    }
+
+    /// The number of focus targets.
+    pub fn len(&self) -> usize {
+        self.items.len()
+    }
+
+    /// Whether the ring holds no targets.
+    pub fn is_empty(&self) -> bool {
+        self.items.is_empty()
+    }
+
+    /// All targets in cycle order.
+    pub fn items(&self) -> &[T] {
+        &self.items
+    }
+
+    /// The currently focused target, if any.
+    pub fn current(&self) -> Option<&T> {
+        self.items.get(self.index?)
+    }
+
+    /// The focused position, if any.
+    pub fn index(&self) -> Option<usize> {
+        self.index
+    }
+
+    /// Whether `item` is the focused target.
+    ///
+    /// Note the empty-ring edge: `is_focused` on any item is `false`.
+    pub fn is_focused(&self, item: &T) -> bool
+    where
+        T: PartialEq,
+    {
+        self.current().is_some_and(|current| current == item)
+    }
+
+    /// Focuses the target at `index`; out-of-range indices are rejected
+    /// and leave the focus unchanged (`false`).
+    pub fn set_index(&mut self, index: usize) -> bool {
+        if index >= self.items.len() {
+            return false;
+        }
+        self.index = Some(index);
+        true
+    }
+
+    /// Focuses the first target equal to `item` (`false` when absent).
+    pub fn set(&mut self, item: &T) -> bool
+    where
+        T: PartialEq,
+    {
+        let found = self.items.iter().position(|candidate| candidate == item);
+        match found {
+            Some(index) => {
+                self.index = Some(index);
+                true
+            }
+            None => false,
+        }
+    }
+
+    /// Advances to the next target, wrapping around; an empty or
+    /// unfocused ring restarts at the beginning. Returns the new target.
+    pub fn advance(&mut self) -> Option<&T> {
+        if self.items.is_empty() {
+            return None;
+        }
+        self.index = Some(match self.index {
+            Some(index) => (index + 1) % self.items.len(),
+            None => 0,
+        });
+        self.items.get(self.index?)
+    }
+
+    /// Steps back to the previous target, wrapping around; an empty or
+    /// unfocused ring restarts at the end. Returns the new target.
+    pub fn retreat(&mut self) -> Option<&T> {
+        if self.items.is_empty() {
+            return None;
+        }
+        self.index = Some(match self.index {
+            Some(index) => (index + self.items.len() - 1) % self.items.len(),
+            None => self.items.len() - 1,
+        });
+        self.items.get(self.index?)
+    }
+
+    /// Clears the focus; [`FocusRing::current`] becomes `None` until the
+    /// next cycle or [`FocusRing::set`] call.
+    pub fn unfocus(&mut self) {
+        self.index = None;
+    }
+}

+//! The keymap: declarative key bindings with multi-key chords.
+//!
+//! tuvix widget states stay key-agnostic — the app translates events into
+//! state methods. For apps with more than a handful of bindings, the
+//! hand-rolled `match` becomes the app's largest input handler. [`Keymap`]
+//! is that translation as a type: bind sequences of [`KeyPattern`]s to
+//! commands, feed it key events, and it answers [`Matched`](KeymapOutcome::Matched)
+//! (run this command), [`Pending`](KeymapOutcome::Pending) (the event
+//! extended a chord prefix — `g` waiting for a second `g`) or
+//! [`Miss`](KeymapOutcome::Miss) (no binding, matcher reset).
+//!
+//! Matching semantics the tests pin down:
+//!
+//! - only press events advance the matcher; repeats and releases neither
+//!   extend nor reset an open chord;
+//! - a pattern's modifiers are a *subset* of the event's — a `Ctrl+C`
+//!   binding fires for `Ctrl+Alt+C` too, and a bare `c` binding fires for
+//!   `Shift+c` when the event still carries `Char('c')` (terminals report
+//!   shifted letters as the uppercase char, which is a different pattern);
+//! - a full binding that is a proper prefix of a longer binding is
+//!   shadowed by it: with both `g` and `gg` bound, a lone `g` stays
+//!   `Pending` forever. Bind the shorter form on a different key;
+//! - a `Miss` or a `Matched` resets the matcher; a `Pending` keeps it open
+//!   (there is no timeout — reset explicitly or feed the next key).
+//!
+//! # Examples
+//!
+//! ```
+//! use tuvix_core::event::{KeyCode, KeyEvent, KeyModifiers};
+//! use tuvix_core::keymap::{KeyPattern, Keymap, KeymapOutcome};
+//!
+//! #[derive(Debug, Clone, Copy, PartialEq, Eq)]
+//! enum Command {
+//!     Quit,
+//!     GotoTop,
+//!     DeleteWord,
+//! }
+//!
+//! let mut keymap = Keymap::new();
+//! keymap.bind(&[KeyPattern::new(KeyCode::Char('c')).ctrl()], Command::Quit);
+//! keymap.bind(&[KeyPattern::new(KeyCode::Char('g')), KeyPattern::new(KeyCode::Char('g'))], Command::GotoTop);
+//! keymap.bind(&[KeyPattern::new(KeyCode::Char('d')).ctrl(), KeyPattern::new(KeyCode::Backspace)], Command::DeleteWord);
+//!
+//! // A single key resolves immediately.
+//! let quit = KeyEvent::new(KeyCode::Char('c'), KeyModifiers::CONTROL | KeyModifiers::ALT);
+//! assert_eq!(keymap.step(&quit), KeymapOutcome::Matched(Command::Quit));
+//!
+//! // A chord opens with the first key and resolves with the last.
+//! let g = KeyEvent::new(KeyCode::Char('g'), KeyModifiers::empty());
+//! assert_eq!(keymap.step(&g), KeymapOutcome::Pending);
+//! assert!(keymap.is_pending());
+//! assert_eq!(keymap.step(&g), KeymapOutcome::Matched(Command::GotoTop));
+//! assert!(!keymap.is_pending());
+//! ```
+
+use crate::event::{KeyCode, KeyEvent, KeyEventKind, KeyModifiers};
+
+/// One key step in a binding: a key code plus the modifiers it requires.
+///
+/// The pattern matches a [`KeyEvent`] when the codes are equal, the event
+/// is a press, and every modifier the pattern names is held (the event may
+/// hold more).
+///
+/// # Examples
+///
+/// ```
+/// use tuvix_core::event::{KeyCode, KeyEvent, KeyModifiers};
+/// use tuvix_core::keymap::KeyPattern;
+///
+/// let pattern = KeyPattern::new(KeyCode::Char('d')).ctrl();
+/// let event = KeyEvent::new(KeyCode::Char('d'), KeyModifiers::CONTROL);
+/// assert!(pattern.matches(&event));
+///
+/// let plain = KeyEvent::new(KeyCode::Char('d'), KeyModifiers::empty());
+/// assert!(!pattern.matches(&plain));
+/// ```
+#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
+pub struct KeyPattern {
+    /// The key code this pattern matches exactly.
+    pub code: KeyCode,
+    /// The modifiers that must be held (subset semantics).
+    pub modifiers: KeyModifiers,
+}
+
+impl KeyPattern {
+    /// A pattern for `code` with no modifier requirements.
+    pub fn new(code: KeyCode) -> Self {
+        Self {
+            code,
+            modifiers: KeyModifiers::empty(),
+        }
+    }
+
+    /// Requires the control modifier.
+    pub fn ctrl(mut self) -> Self {
+        self.modifiers.insert(KeyModifiers::CONTROL);
+        self
+    }
+
+    /// Requires the alt modifier.
+    pub fn alt(mut self) -> Self {
+        self.modifiers.insert(KeyModifiers::ALT);
+        self
+    }
+
+    /// Requires the shift modifier.
+    pub fn shift(mut self) -> Self {
+        self.modifiers.insert(KeyModifiers::SHIFT);
+        self
+    }
+
+    /// Requires the super modifier.
+    pub fn super_key(mut self) -> Self {
+        self.modifiers.insert(KeyModifiers::SUPER);
+        self
+    }
+
+    /// Requires every modifier in `modifiers` (union with the current set).
+    pub fn with(mut self, modifiers: KeyModifiers) -> Self {
+        self.modifiers.insert(modifiers);
+        self
+    }
+
+    /// Whether `key` satisfies this pattern: same code, a press, and the
+    /// pattern's modifiers a subset of the event's.
+    pub fn matches(&self, key: &KeyEvent) -> bool {
+        key.kind == KeyEventKind::Press
+            && key.code == self.code
+            && key.modifiers.contains(self.modifiers)
+    }
+}

+/// What one fed key event means to a [`Keymap`].
+#[derive(Debug, Clone, Copy, PartialEq, Eq)]
+pub enum KeymapOutcome<C> {
+    /// The event completed a binding; the matcher is reset.
+    Matched(C),
+    /// The event extended a chord prefix; the matcher stays open.
+    Pending,
+    /// The event matches nothing; the matcher is reset.
+    Miss,
+}
+
+/// The binding trie: sequences of [`KeyPattern`]s mapped to commands.
+///
+/// The trie keeps the matcher's position between events, so chords like
+/// `g g` resolve across two [`Keymap::step`] calls without the app
+/// carrying any state.
+#[derive(Debug, Clone, PartialEq, Eq)]
+pub struct Keymap<C> {
+    root: Node<C>,
+    path: Vec<usize>,
+}
+
+#[derive(Debug, Clone, PartialEq, Eq)]
+struct Node<C> {
+    children: Vec<(KeyPattern, Node<C>)>,
+    command: Option<C>,
+}
+
+impl<C> Default for Keymap<C> {
+    /// An empty keymap — `C` needs no default of its own.
+    fn default() -> Self {
+        Self {
+            root: Node {
+                children: Vec::new(),
+                command: None,
+            },
+            path: Vec::new(),
+        }
+    }
+}
+
+impl<C: Clone> Keymap<C> {
+    /// Creates an empty keymap.
+    pub fn new() -> Self {
+        Self::default()
+    }
+
+    /// Binds `sequence` to `command`; later bindings of the same sequence
+    /// overwrite earlier ones. An empty sequence cannot be bound (`false`).
+    pub fn bind(&mut self, sequence: &[KeyPattern], command: C) -> bool {
+        if sequence.is_empty() {
+            return false;
+        }
+        let mut path: Vec<usize> = Vec::new();
+        for pattern in sequence {
+            let node = self.node_mut(&path);
+            if let Some(index) = node.children.iter().position(|(known, _)| known == pattern) {
+                path.push(index);
+            } else {
+                node.children.push((
+                    *pattern,
+                    Node {
+                        children: Vec::new(),
+                        command: None,
+                    },
+                ));
+                path.push(node.children.len() - 1);
+            }
+        }
+        self.node_mut(&path).command = Some(command);
+        true
+    }
+
+    /// Feeds one key event to the matcher.
+    ///
+    /// Only press events advance matching; a repeat or release mid-chord
+    /// reports [`KeymapOutcome::Pending`] (the chord stays open) and
+    /// [`KeymapOutcome::Miss`] at the root.
+    pub fn step(&mut self, key: &KeyEvent) -> KeymapOutcome<C> {
+        // Repeats and releases carry no chord information: they neither
+        // extend nor reset an open sequence.
+        if key.kind != KeyEventKind::Press {
+            return if self.path.is_empty() {
+                KeymapOutcome::Miss
+            } else {
+                KeymapOutcome::Pending
+            };
+        }
+
+        let decision = {
+            let node = self.node_at(&self.path);
+            node.children
+                .iter()
+                .position(|(pattern, _)| pattern.matches(key))
+                .map(|index| {
+                    let child = &node.children[index].1;
+                    (index, child.children.is_empty())
+                })
+        };
+
+        let Some((index, leaf)) = decision else {
+            self.path.clear();
+            return KeymapOutcome::Miss;
+        };
+        self.path.push(index);
+        if leaf {
+            let command = self.node_at(&self.path).command.clone();
+            self.path.clear();
+            return match command {
+                Some(command) => KeymapOutcome::Matched(command),
+                // Unreachable by construction (`bind` always plants a
+                // command at the leaf), but the contract stays honest.
+                None => KeymapOutcome::Miss,
+            };
+        }
+        KeymapOutcome::Pending
+    }
+
+    /// Whether a chord prefix is open (the last step was
+    /// [`KeymapOutcome::Pending`]).
+    pub fn is_pending(&self) -> bool {
+        !self.path.is_empty()
+    }
+
+    /// Closes an open chord prefix, forgetting the keys fed so far.
+    pub fn reset(&mut self) {
+        self.path.clear();
+    }
+
+    /// The node reached by walking `path` from the root.
+    fn node_at(&self, path: &[usize]) -> &Node<C> {
+        let mut node = &self.root;
+        for &index in path {
+            node = &node.children[index].1;
+        }
+        node
+    }
+
+    /// The node reached by walking `path` from the root, mutably.
+    fn node_mut(&mut self, path: &[usize]) -> &mut Node<C> {
+        let mut node = &mut self.root;
+        for &index in path {
+            node = &mut node.children[index].1;
+        }
+        node
+    }
+}

+//! Benchmarks for the M11 `Scrollable` viewport: the paint-into-scratch +
+//! blit pipeline. The gate these back is B14 (a scrolled 200x60 repaint
+//! inside the steady-frame budget), measured as an `#[ignore]` timing test
+//! in `tests/m11_interaction.rs`.
+
+use std::hint::black_box;
+
+use criterion::{Criterion, Throughput, criterion_group, criterion_main};
+use tuvix_core::buffer::Buffer;
+use tuvix_core::geometry::{Rect, Size};
+use tuvix_core::style::Style;
+use tuvix_core::widget::RenderWidget;
+use tuvix_widgets::{Scrollable, Scrollbar, ViewportState};
+
+/// The architecture budget viewport: 200x60.
+const WIDTH: u16 = 200;
+const HEIGHT: u16 = 60;
+/// Four hundred rows of realistic log-like content behind the viewport.
+const CONTENT_ROWS: u16 = 400;
+
+/// A log-like content widget: one line per row, rendered like a real app.
+#[derive(Clone, Copy)]
+struct LogLines(u16);
+
+impl RenderWidget for LogLines {
+    type State = ();
+
+    fn render(self, area: Rect, buf: &mut Buffer, _state: &mut ()) {
+        for row in 0..self.0.min(area.height) {
+            let line = format!(
+                "2026-11-05 12:{row:02}:00 INFO  worker-{row} handled request in {ms}ms",
+                ms = row % 97
+            );
+            buf.set_string(area.x, area.y + row, line, Style::new());
+        }
+    }
+}
+
+fn bench_scrollable(c: &mut Criterion) {
+    let mut group = c.benchmark_group("scrollable");
+    group.throughput(Throughput::Elements(u64::from(WIDTH) * u64::from(HEIGHT)));
+    let area = Rect::new(0, 0, WIDTH, HEIGHT);
+
+    // First paint: the scratch surface starts empty and the child paints
+    // the full 400-row content once.
+    group.bench_function("first_paint_400_rows_200x60", |b| {
+        b.iter_batched(
+            || {
+                (
+                    Scrollable::new(LogLines(CONTENT_ROWS)).content(Size::new(WIDTH, CONTENT_ROWS)),
+                    ViewportState::new(()),
+                )
+            },
+            |(widget, mut state)| {
+                widget
+                    .clone()
+                    .render(black_box(area), &mut Buffer::empty(area), &mut state);
+            },
+            criterion::BatchSize::SmallInput,
+        );
+    });
+
+    // Steady state: the scratch surface is warm; a scroll step blits the
+    // new window out of the cached paint. This is the B14 cost.
+    group.bench_function("scrolled_repaint_400_rows_200x60", |b| {
+        let widget =
+            Scrollable::new(LogLines(CONTENT_ROWS)).content(Size::new(WIDTH, CONTENT_ROWS));
+        let mut state = ViewportState::new(());
+        widget
+            .clone()
+            .render(area, &mut Buffer::empty(area), &mut state);
+        b.iter(|| {
+            state.scroll_down(1);
+            widget
+                .clone()
+                .render(black_box(area), &mut Buffer::empty(area), &mut state);
+        });
+    });
+
+    // After an invalidation: the child paints the full content again and
+    // the window blits — the app-paid cost of a content change.
+    group.bench_function("invalidated_repaint_400_rows_200x60", |b| {
+        let widget =
+            Scrollable::new(LogLines(CONTENT_ROWS)).content(Size::new(WIDTH, CONTENT_ROWS));

+                .clone()

+//! Scrollable viewports: clip any widget to a window over larger content.
+//!
+//! A [`Scrollable`] wraps any [`RenderWidget`], paints it into an offscreen
+//! scratch surface sized to the declared content extent, and blits the
+//! window the [`ViewportState`] currently scrolls over. The scratch surface
+//! lives in the state and the child is painted **once** — between
+//! [`ViewportState::invalidate`] calls the frame is a pure blit, so a
+//! scroll storm costs the window, never the content. Every write goes
+//! through the journal-recorded
+//! [`Buffer::get_mut`](tuvix_core::buffer::Buffer::get_mut), so the
+//! persistent-DOM diff still sees exactly the blitted region.
+//!
+//! Semantics the tests pin down:
+//!
+//! - offsets are clamped to `content − viewport` at render, so a resize can
+//!   never leave the window outside the content;
+//! - a wide glyph whose pair would be split by the right window edge is
+//!   clipped (blanked), never emitted as a lone leading half — the same
+//!   terminal-pair honesty the buffer's write path enforces;
+//! - a window starting on a covered half snaps back one column to the
+//!   leading edge, so horizontal scrolling always starts on a whole glyph;
+//! - wheel events move one line or column per notch — a primitive the app
+//!   composes, not a policy.
+
+use tuvix_core::buffer::{Buffer, Cell};
+use tuvix_core::event::{MouseEvent, MouseEventKind};
+use tuvix_core::geometry::{Rect, Size};
+use tuvix_core::widget::RenderWidget;
+
+use crate::scrollbar::{Scrollbar, ScrollbarState};
+
+/// The scroll state of a [`Scrollable`]: the window offsets plus the
+/// offscreen scratch surface the inner widget paints into.
+///
+/// The child's paint is cached: a frame where nothing changed but the
+/// window blits, a frame where the content changed needs
+/// [`ViewportState::invalidate`] first — the same explicit-dirtiness
+/// contract the persistent-DOM journal uses one level down.
+///
+/// The state also remembers the last rendered viewport and the declared
+/// content extent, so the scroll operations clamp honestly. Rendering
+/// syncs both; before the first frame the app can prime them with
+/// [`ViewportState::set_content`] (the viewport starts at `0 x 0`, so
+/// before priming the clamps derive from the content alone).
+///
+/// The inner widget's own state is carried along — the app reaches it
+/// through [`ViewportState::inner`] and [`ViewportState::inner_mut`].
+///
+/// # Examples
+///
+/// ```
+/// use tuvix_core::geometry::Size;
+/// use tuvix_widgets::ViewportState;
+///
+/// let mut state = ViewportState::new(());
+/// state.set_content(Size::new(40, 100));
+/// assert!(state.scroll_down(30));
+/// // The viewport is still 0 x 0, so the clamp is the content height:
+/// assert_eq!(state.offset_y(), 30);
+/// ```
+#[derive(Debug, Clone, PartialEq, Eq)]
+pub struct ViewportState<S> {
+    offset_y: u16,
+    offset_x: u16,
+    content: Size,
+    viewport: Size,
+    scratch: Option<(Size, Buffer)>,
+    valid: bool,
+    inner: S,
+}
+
+impl<S> ViewportState<S> {
+    /// Creates the state around the inner widget's own state.
+    pub fn new(inner: S) -> Self {
+        Self {
+            offset_y: 0,
+            offset_x: 0,
+            content: Size::new(0, 0),
+            viewport: Size::new(0, 0),
+            scratch: None,
+            valid: false,
+            inner,
+        }
+    }
+
+    /// The vertical scroll offset in content rows.
+    pub fn offset_y(&self) -> u16 {
+        self.offset_y
+    }
+
+    /// The horizontal scroll offset in content columns.
+    pub fn offset_x(&self) -> u16 {
+        self.offset_x
+    }
+
+    /// The last rendered viewport size (`0 x 0` before the first frame).
+    pub fn viewport(&self) -> Size {
+        self.viewport
+    }
+
+    /// The declared content extent this state clamps against.
+    pub fn content(&self) -> Size {
+        self.content
+    }
+
+    /// Declares the content extent the scroll offsets clamp against.
+    ///
+    /// Rendering syncs this from the widget; priming it up front makes the
+    /// scroll operations honest before the first frame.
+    pub fn set_content(&mut self, content: Size) -> &mut Self {
+        self.content = content;
+        self.clamp_offsets();
+        self
+    }
+
+    /// Marks the child's paint stale: the next frame re-paints the scratch
+    /// surface before blitting.
+    ///
+    /// Call this whenever anything the inner widget reads changes — its
+    /// state through [`ViewportState::inner_mut`], or the data it captured
+    /// by value. A frame that only scrolls needs no invalidation.
+    pub fn invalidate(&mut self) -> &mut Self {
+        self.valid = false;
+        self
+    }
+
+    /// Whether the cached child paint is current (no invalidation pending
+    /// and the scratch surface exists at the right size).
+    pub fn is_valid(&self) -> bool {
+        self.valid
+    }
+
+    /// The inner widget's state.
+    pub fn inner(&self) -> &S {
+        &self.inner
+    }
+
+    /// The inner widget's state, mutably.
+    pub fn inner_mut(&mut self) -> &mut S {
+        &mut self.inner
+    }
+
+    /// The largest legal vertical offset for the current content/viewport.
+    fn max_offset_y(&self) -> u16 {
+        self.content.height.saturating_sub(self.viewport.height)
+    }
+
+    /// The largest legal horizontal offset for the current content/viewport.
+    fn max_offset_x(&self) -> u16 {
+        self.content.width.saturating_sub(self.viewport.width)
+    }
+
+    fn clamp_offsets(&mut self) {
+        self.offset_y = self.offset_y.min(self.max_offset_y());
+        self.offset_x = self.offset_x.min(self.max_offset_x());
+    }
+
+    fn shift_y(&mut self, delta: i32) -> bool {
+        self.clamp_offsets();
+        let current = i32::from(self.offset_y);
+        let target = if delta >= 0 {
+            current
+                .saturating_add(delta)
+                .min(i32::from(self.max_offset_y()))
+        } else {
+            current.saturating_add(delta).max(0)
+        };
+        let target = target.max(0) as u16;
+        if target == self.offset_y {
+            return false;
+        }
+        self.offset_y = target;
+        true
+    }
+
+    fn shift_x(&mut self, delta: i32) -> bool {
+        self.clamp_offsets();
+        let current = i32::from(self.offset_x);
+        let target = if delta >= 0 {
+            current
+                .saturating_add(delta)
+                .min(i32::from(self.max_offset_x()))
+        } else {
+            current.saturating_add(delta).max(0)
+        };
+        let target = target.max(0) as u16;
+        if target == self.offset_x {
+            return false;
+        }
+        self.offset_x = target;
+        true
+    }
+
+    /// Scrolls up by `rows` content rows; `true` when the offset changed.
+    pub fn scroll_up(&mut self, rows: u16) -> bool {
+        self.shift_y(-i32::from(rows))
+    }
+
+    /// Scrolls down by `rows` content rows; `true` when the offset changed.
+    pub fn scroll_down(&mut self, rows: u16) -> bool {
+        self.shift_y(i32::from(rows))
+    }
+
+    /// Scrolls left by `columns`; `true` when the offset changed.
+    pub fn scroll_left(&mut self, columns: u16) -> bool {
+        self.shift_x(-i32::from(columns))
+    }
+
+    /// Scrolls right by `columns`; `true` when the offset changed.
+    pub fn scroll_right(&mut self, columns: u16) -> bool {
+        self.shift_x(i32::from(columns))
+    }
+
+    /// Pages up by `viewport_height` rows; `true` when the offset changed.
+    pub fn page_up(&mut self, viewport_height: u16) -> bool {
+        self.shift_y(-i32::from(viewport_height))
+    }
+
+    /// Pages down by `viewport_height` rows; `true` when the offset changed.
+    pub fn page_down(&mut self, viewport_height: u16) -> bool {
+        self.shift_y(i32::from(viewport_height))
+    }
+
+    /// Jumps to the top of the content; `true` when the offset changed.
+    pub fn to_top(&mut self) -> bool {
+        self.shift_y(-i32::MAX)
+    }
+
+    /// Jumps to the bottom of the content; `true` when the offset changed.
+    pub fn to_bottom(&mut self) -> bool {
+        self.shift_y(i32::MAX)
+    }
+
+    /// Jumps to the left edge of the content; `true` when the offset changed.
+    pub fn to_left_end(&mut self) -> bool {
+        self.shift_x(-i32::MAX)
+    }
+
+    /// Jumps to the right edge of the content; `true` when the offset changed.
+    pub fn to_right_end(&mut self) -> bool {
+        self.shift_x(i32::MAX)
+    }
+
+    /// Applies a mouse wheel event: one row or column per wheel notch.
+    ///
+    /// All other mouse events are ignored (`false`).
+    pub fn apply_wheel(&mut self, event: &MouseEvent) -> bool {
+        match event.kind {
+            MouseEventKind::ScrollUp => self.scroll_up(1),
+            MouseEventKind::ScrollDown => self.scroll_down(1),
+            MouseEventKind::ScrollLeft => self.scroll_left(1),
+            MouseEventKind::ScrollRight => self.scroll_right(1),
+            _ => false,
+        }
+    }
+}
+
+impl<S: Default> Default for ViewportState<S> {
+    fn default() -> Self {
+        Self::new(S::default())
+    }
+}

+/// A viewport over larger content: any [`RenderWidget`] plus a declared
+/// content extent, optionally with an attached vertical [`Scrollbar`].
+///
+/// The inner widget is painted at its full declared size into the state's
+/// scratch surface, then the window [`ViewportState`] scrolls over is
+/// blitted into the real buffer. When the content is taller than the
+/// viewport and a scrollbar is attached, the scrollbar takes the last
+/// column and the viewport shrinks by it.
+///
+/// # Examples
+///
+/// ```
+/// use tuvix_core::buffer::Buffer;
+/// use tuvix_core::geometry::{Rect, Size};
+/// use tuvix_core::style::Style;
+/// use tuvix_core::widget::RenderWidget;
+/// use tuvix_widgets::{Scrollable, ViewportState};
+///
+/// struct Lines<'a>(&'a [&'a str]);
+///
+/// impl RenderWidget for Lines<'_> {
+///     type State = ();
+///     fn render(self, area: Rect, buf: &mut Buffer, _state: &mut ()) {
+///         for (index, line) in self.0.iter().enumerate() {
+///             buf.set_string(area.x, area.y + index as u16, *line, Style::new());
+///         }
+///     }
+/// }
+///
+/// let content = ["one", "two", "three", "four"];
+/// let mut buffer = Buffer::empty(Rect::new(0, 0, 5, 2));
+/// let mut state = ViewportState::new(());
+/// Scrollable::new(Lines(&content))
+///     .content(Size::new(5, 4))
+///     .render(Rect::new(0, 0, 5, 2), &mut buffer, &mut state);
+/// assert_eq!(buffer.to_string(), "one  \ntwo  ");
+///
+/// state.scroll_down(2);
+/// Scrollable::new(Lines(&content))
+///     .content(Size::new(5, 4))
+///     .render(Rect::new(0, 0, 5, 2), &mut buffer, &mut state);
+/// assert_eq!(buffer.to_string(), "three\nfour ");
+/// ```
+#[derive(Debug, Clone, PartialEq, Eq)]
+pub struct Scrollable<W> {
+    inner: W,
+    content: Size,
+    scrollbar: Option<Scrollbar>,
+}
+
+impl<W> Scrollable<W> {
+    /// Wraps `inner`, whose painted extent is declared by `.content(...)`.
+    ///
+    /// Without a declared content extent the viewport cannot scroll
+    /// (the clamps derive from `content − viewport`).
+    pub fn new(inner: W) -> Self {
+        Self {
+            inner,
+            content: Size::new(0, 0),
+            scrollbar: None,
+        }
+    }
+
+    /// Declares the painted extent of the inner widget.
+    pub fn content(mut self, content: Size) -> Self {
+        self.content = content;
+        self
+    }
+
+    /// Attaches a vertical scrollbar (rendered in the last column when the
+    /// content is taller than the viewport).
+    pub fn scrollbar(mut self, scrollbar: Scrollbar) -> Self {
+        self.scrollbar = Some(scrollbar);
+        self
+    }
+
+    /// The declared content extent.
+    pub fn get_content(&self) -> Size {
+        self.content
+    }
+
+    /// The attached scrollbar, if any.
+    pub fn get_scrollbar(&self) -> Option<&Scrollbar> {
+        self.scrollbar.as_ref()
+    }
+}
+
+impl<W: RenderWidget> RenderWidget for Scrollable<W> {
+    type State = ViewportState<W::State>;
+
+    fn render(self, area: Rect, buf: &mut Buffer, state: &mut Self::State) {
+        if area.is_empty() {
+            return;
+        }
+
+        // The scrollbar claims the last column only when it can help
+        // (attached, content taller than the window, room for both).
+        let scroll_slot = self
+            .scrollbar
+            .filter(|_| self.content.height > area.height)
+            .filter(|_| area.width > 1);
+        let (viewport_area, scrollbar) = match scroll_slot {
+            Some(scrollbar) => {
+                let column = Rect::new(area.right() - 1, area.y, 1, area.height);
+                let viewport = Rect::new(area.x, area.y, area.width - 1, area.height);
+                (viewport, Some((scrollbar, column)))
+            }
+            None => (area, None),
+        };
+
+        // Sync the clamping world and enforce it.
+        state.content = self.content;
+        state.viewport = viewport_area.size();
+        state.clamp_offsets();
+
+        // Paint the inner widget at its full extent into the scratch
+        // surface — once per invalidation, not once per frame. A frame
+        // that only scrolls blits the cached paint.
+        let paint = Rect::new(
+            0,
+            0,
+            self.content.width.max(viewport_area.width),
+            self.content.height.max(viewport_area.height),
+        );
+        let needs_paint = !state.valid
+            || state
+                .scratch
+                .as_ref()
+                .is_none_or(|(size, _)| *size != paint.size());
+        state
+            .scratch
+            .get_or_insert_with(|| (paint.size(), Buffer::empty(paint)));
+        if needs_paint {
+            let (size, scratch) = state.scratch.as_mut().expect("just inserted");
+            *size = paint.size();
+            scratch.resize(paint);
+            scratch.reset();
+            self.inner.render(paint, scratch, &mut state.inner);
+        }
+        state.valid = true;
+        let scratch = &state.scratch.as_ref().expect("just inserted").1;
+
+        // Snap a window starting on a covered half back to the leading
+        // edge so horizontal scrolling always begins on a whole glyph.
+        let mut offset_x = state.offset_x;
+        if offset_x > 0
+            && scratch
+                .get(offset_x, state.offset_y)
+                .is_some_and(|cell| cell.skip)
+        {
+            offset_x -= 1;
+        }
+
+        let window = Rect::new(
+            offset_x,
+            state.offset_y,
+            viewport_area.width,
+            viewport_area.height,
+        );
+        for row in 0..viewport_area.height {
+            let y = window.y + row;
+            for column in 0..viewport_area.width {
+                let x = window.x + column;
+                let Some(cell) = scratch.get(x, y) else {
+                    continue;
+                };
+                let mut cell = cell.clone();
+                // A wide glyph whose covered half falls outside the window
+                // would arrive as a lone leading half mid-row: clip it.
+                if !cell.skip
+                    && column + 1 == viewport_area.width
+                    && scratch.get(x + 1, y).is_some_and(|next| next.skip)
+                {
+                    cell = Cell::empty();
+                }
+                if let Some(target) = buf.get_mut(viewport_area.x + column, viewport_area.y + row) {
+                    *target = cell;
+                }
+            }
+        }
+
+        if let Some((scrollbar, column)) = scrollbar {
+            let mut thumb = ScrollbarState::new(usize::from(self.content.height));
+            thumb
+                .position(usize::from(state.offset_y))
+                .viewport_content_length(usize::from(viewport_area.height));
+            scrollbar.render(column, buf, &mut thumb);
+        }
+    }
+}

+//! M11 integration gates: the B14 viewport-scroll budget, measured, not
+//! asserted.
+//!
+//! B14 — a scrolled 200x60 viewport frame over 400 rows of content (the
+//! window moves, the cached child paint blits) stays inside **1 ms**, half
+//! of the B2 steady-frame budget, so a scroll storm can never push a frame
+//! over the 2 ms line by itself. A content change additionally pays the
+//! child's own paint when the app invalidates — that cost is the child's,
+//! not the viewport's (measured in the `scrollable` bench group).
+
+use std::hint::black_box;
+use std::time::{Duration, Instant};
+
+use tuvix_core::buffer::Buffer;
+use tuvix_core::geometry::{Rect, Size};
+use tuvix_core::style::Style;
+use tuvix_core::widget::RenderWidget;
+use tuvix_widgets::{Scrollable, ViewportState};
+
+const WIDTH: u16 = 200;
+const HEIGHT: u16 = 60;
+const CONTENT_ROWS: u16 = 400;
+const FRAMES: usize = 200;
+
+#[derive(Clone, Copy)]
+struct LogLines(u16);
+
+impl RenderWidget for LogLines {
+    type State = ();
+
+    fn render(self, area: Rect, buf: &mut Buffer, _state: &mut ()) {
+        for row in 0..self.0.min(area.height) {
+            let line = format!(
+                "2026-11-05 12:{row:02}:00 INFO  worker-{row} handled request in {ms}ms",
+                ms = row % 97
+            );
+            buf.set_string(area.x, area.y + row, line, Style::new());
+        }
+    }
+}

+/// One scroll step: blit the window out of the cached child paint —
+/// exactly what a scroll-storm frame costs.
+fn scroll_frame(
+    widget: &Scrollable<LogLines>,
+    state: &mut ViewportState<()>,
+    area: Rect,
+) -> Buffer {
+    state.scroll_down(1);
+    let mut buffer = Buffer::empty(area);
+    widget.clone().render(area, &mut buffer, state);
+    buffer
+}

+#[test]
+#[ignore = "timing gate: run with --ignored (perf workflow)"]
+fn b14_viewport_scroll_repaint_within_one_millisecond() {
+    let area = Rect::new(0, 0, WIDTH, HEIGHT);
+    let widget = Scrollable::new(LogLines(CONTENT_ROWS)).content(Size::new(WIDTH, CONTENT_ROWS));
+    let mut state = ViewportState::new(());
+
+    // Warm the scratch surface (the first paint is not the gate).
+    widget
+        .clone()
+        .render(area, &mut Buffer::empty(area), &mut state);
+
+    let mut latencies = Vec::with_capacity(FRAMES);
+    for frame in 0..FRAMES {
+        let start = Instant::now();
+        let buffer = black_box(scroll_frame(&widget, black_box(&mut state), area));
+        let elapsed = start.elapsed();
+        // Sanity: the window really scrolled (content moved by the frame
+        // count).
+        if frame < usize::from(CONTENT_ROWS - HEIGHT - 1) {
+            assert!(
+                buffer.to_string().contains(&format!("12:{:02}", frame + 1)),
+                "the scrolled window must show fresh rows"
+            );
+        }
+        latencies.push(elapsed);
+    }
+    latencies.sort();
+    let median = latencies[FRAMES / 2];
+    let p99 = latencies[FRAMES * 99 / 100 - 1];
+    let max = latencies[FRAMES - 1];
+
+    assert!(
+        p99 <= Duration::from_millis(1),
+        "B14 gate failed: viewport scroll repaint p99 {p99:?} (budget 1 ms)"
+    );
+    println!(
+        "B14 gate: viewport scroll repaint median {median:?}, p99 {p99:?}, max {max:?} over {FRAMES} frames @ {WIDTH}x{HEIGHT}, {CONTENT_ROWS} content rows (budget 1 ms)"
+    );
+}

+//! and the `Keymap` working as one app. Keys:
+//! - `Tab`/`Shift+Tab` move the focus ring (log pane, name, filter),

+//! - typing edits the focused input.

+fn main_app() -> Result<()> {
+    let mut runner = try_init()?;

+    let (mut reader, mut events) = CrosstermEvents::spawn()?;
+    let keymap = dashboard_keymap();

+    let content_height = lines.len() as u16;

+/// Event pump: the keymap resolves first; whatever it misses feeds the
+/// focused pane.
+fn pump_events(app: &mut App, events: &mut tuvix_core::runtime::queue::EventStream) {
+    let deadline = Instant::now() + Duration::from_millis(120);
+    while Instant::now() < deadline {
+        match events.try_next() {
+            Some(Event::Key(key)) => match app.keymap.step(&key) {
+                KeymapOutcome::Matched(command) => dispatch(app, command),
+                KeymapOutcome::Pending => {}
+                KeymapOutcome::Miss => {
+                    // The keymap missed: editing keys feed the focused pane.
+                    if key.kind == KeyEventKind::Press {
+                        apply_missed(app, key);
+                    }
+                }
+            },
+            Some(Event::Mouse(mouse)) => {
+                if app.focus.is_focused(&Pane::Log) {
+                    app.viewport.apply_wheel(&mouse);
+                }
+            }
+            Some(_) => {}
+            None => break,
+        }
+    }
+}
+
+fn dispatch(app: &mut App, command: Command) {
+    match command {
+        Command::Quit => app.quit = true,
+        Command::FocusNext => {
+            app.focus.advance();
+        }
+        Command::FocusPrevious => {
+            app.focus.retreat();
+        }
+        Command::ToTop => {
+            app.viewport.to_top();
+        }
+        Command::ToBottom => {
+            app.viewport.to_bottom();
+        }
+        Command::PageDown => {
+            let page = app.viewport.viewport().height.max(1);
+            app.viewport.scroll_down(page);
+        }
+        Command::PageUp => {
+            let page = app.viewport.viewport().height.max(1);
+            app.viewport.scroll_up(page);
+        }
+    }
+}
+
+/// Keys the keymap missed: scroll keys for the log pane, editing keys for
+/// the inputs.
+fn apply_missed(app: &mut App, key: KeyEvent) {
+    let modifiers = key.modifiers;
+    match key.code {
+        KeyCode::Down | KeyCode::Char('j') => {
+            if app.focus.is_focused(&Pane::Log) {
+                app.viewport.scroll_down(1);
+            }
+        }
+        KeyCode::Up | KeyCode::Char('k') => {
+            if app.focus.is_focused(&Pane::Log) {
+                app.viewport.scroll_up(1);
+            }
+        }
+        _ => match app.focus.current() {
+            Some(Pane::Name) => edit(&mut app.name, key, modifiers),
+            Some(Pane::Filter) => edit(&mut app.filter, key, modifiers),
+            _ => {}
+        },
+    }
+}
+
+fn edit(state: &mut InputState, key: KeyEvent, modifiers: KeyModifiers) {
+    match key.code {
+        KeyCode::Backspace => state.delete_backward(),
+        KeyCode::Delete => state.delete_forward(),
+        KeyCode::Left if modifiers.contains(KeyModifiers::CONTROL) => state.move_word_left(),
+        KeyCode::Right if modifiers.contains(KeyModifiers::CONTROL) => state.move_word_right(),
+        KeyCode::Left => state.move_left(),
+        KeyCode::Right => state.move_right(),
+        KeyCode::Home => state.move_home(),
+        KeyCode::End => state.move_end(),
+        KeyCode::Char(character) => state.insert(character),
+        _ => {}
+    }
+}

+                row % 5,
+                1000 + row,
+                (row * 13) % 120
+            )
+        })

+## 18. M11 decisions (2026-11)
+
+- **The viewport's scratch surface lives in the state and is invalidated
+  explicitly, not re-rendered every frame.** The first implementation
+  re-painted the child on every frame and the B14 gate failed it honestly:
+  a 400-row content repaint pushed the scroll step to ~1.7 ms, over the
+  1 ms budget, before any diff ran. The shipped design paints the child
+  once per invalidation and blits the window the rest of the time —
+  ~166 us per scroll step, the same explicit-dirtiness contract the
+  persistent-DOM journal uses one level down. A frame that only scrolls
+  costs the window; a content change costs the child's own paint on top,
+  which is the child's cost, not the viewport's.
+- **The scroll state remembers the world it clamps against.** Offsets are
+  clamped to `content − viewport` at render, and the state keeps both
+  numbers so the scroll operations are honest between frames too
+  (`scroll_down` returns `false` at the bottom instead of lying until the
+  next render). Before the first frame the app can prime the content
+  extent with `set_content`; the zero viewport then means the clamp is the
+  bare content, which is what "nothing is visible yet" means.
+- **Wide glyphs are never split by the window edge — in either direction.**
+  A pair crossing the *right* edge is clipped (blanked), never emitted as
+  a lone leading half mid-row, and a window starting on a covered half
+  snaps back one column to the leading edge. The same terminal-pair
+  honesty the buffer's write path enforces, applied at the viewport
+  boundary; the unit tests pin both edges over CJK content.
+- **Focus is routing, not a widget trait.** Widget states stay
+  key-agnostic (the M6 contract), so focus is the app's mapping question:
+  "which pane receives the editing keys?" `FocusRing<T>` answers it over
+  the app's own pane enum — wrap-around `advance`/`retreat`, lookup by
+  value or index, an explicit unfocused state, and no trait for widgets to
+  implement. The methods are `advance`/`retreat`, not `next`/`previous`:
+  a ring is not an `Iterator` (no termination, returns a reference), and
+  confusing the two names buys nothing.
+- **The keymap shadows prefixes, on purpose.** With both `g` and `gg`
+  bound, a lone `g` stays `Pending` forever — the trie cannot know a chord
+  is over without a timeout, and a timeout is a policy the framework
+  should not own (`reset` exists for apps that want one). Modifiers match
+  as a subset (a `Ctrl+C` binding fires for `Ctrl+Alt+C`), only press
+  events advance matching, and commands are cloned out so `step` needs no
+  lifetime on the outcome — the app owns a plain `Matched(Command)`.
+  `Miss` and `Matched` reset the matcher; `Pending` keeps it open.
+- **Zero new dependencies, again.** The trie is a `Vec`-backed tree, the
+  ring a `Vec` and an index, the viewport a second `Buffer`. The M5
+  framing argument still holds: the fewer bytes that cross the trust
+  boundary, the fewer the escape hatches to audit.
